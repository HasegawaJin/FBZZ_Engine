# Renderer / DX11 Implementation

DX11 具体実装。`engine/src/Renderer/Platform/DX11/` に配置し、上位レイヤーからは隠蔽する。

---

## クラス一覧

| クラス | 基底 | 役割 |
|--------|------|------|
| `DX11Renderer` | `IRenderer` | デバイス・スワップチェーン管理、描画実行 |
| `DX11Buffer` | `IBuffer` | 頂点・インデックスバッファ |
| `DX11Shader` | `IShader` | VS/PS コンパイル・バインド |
| `DX11Texture` | `ITexture` | テクスチャ読み込み・SRV 管理 |

---

## DX11Renderer

```cpp
namespace fbzz::renderer {

class DX11Renderer : public IRenderer {
public:
    // ウィンドウハンドルを受け取って初期化
    bool Init(HWND hwnd, uint32_t width, uint32_t height);
    void Shutdown();

    void BeginFrame() override;
    void EndFrame()   override;
    void Clear(const math::Vec4& color) override;

    std::shared_ptr<IBuffer>  CreateVertexBuffer(const void* data, size_t sizeBytes) override;
    std::shared_ptr<IBuffer>  CreateIndexBuffer(const void* data, uint32_t count) override;
    std::shared_ptr<IShader>  CreateShader(const std::string& path) override;
    std::shared_ptr<ITexture> CreateTexture(const std::string& path) override;

    void Submit(const DrawCall& call) override;
    void DrawLine(const math::Vec3& from, const math::Vec3& to,
                  const math::Vec4& color) override;
    void DrawSphere(const math::Vec3& center, float radius,
                    const math::Vec4& color) override;
    void DrawAABB(const math::Vec3& min, const math::Vec3& max,
                  const math::Vec4& color) override;

    // 内部アクセス (DX11Buffer 等が使う)
    ID3D11Device*        GetDevice()        const { return m_device.Get(); }
    ID3D11DeviceContext* GetDeviceContext()  const { return m_context.Get(); }

private:
    Microsoft::WRL::ComPtr<ID3D11Device>           m_device;
    Microsoft::WRL::ComPtr<ID3D11DeviceContext>    m_context;
    Microsoft::WRL::ComPtr<IDXGISwapChain>         m_swapChain;
    Microsoft::WRL::ComPtr<ID3D11RenderTargetView> m_renderTargetView;
    Microsoft::WRL::ComPtr<ID3D11DepthStencilView> m_depthStencilView;
    Microsoft::WRL::ComPtr<ID3D11Texture2D>        m_depthStencilBuffer;

    uint32_t m_width  = 0;
    uint32_t m_height = 0;
};

} // namespace fbzz::renderer
```

## DX11Buffer

```cpp
namespace fbzz::renderer {

class DX11Buffer : public IBuffer {
public:
    bool Init(ID3D11Device* device, const void* data,
              size_t sizeBytes, uint32_t stride, D3D11_BIND_FLAG bindFlag);

    void Update(const void* data, size_t sizeBytes) override;
    size_t   GetSize()   const override { return m_size; }
    uint32_t GetStride() const override { return m_stride; }

    ID3D11Buffer* GetBuffer() const { return m_buffer.Get(); }

private:
    Microsoft::WRL::ComPtr<ID3D11Buffer> m_buffer;
    ID3D11DeviceContext* m_context = nullptr;
    size_t   m_size   = 0;
    uint32_t m_stride = 0;
};

} // namespace fbzz::renderer
```

## DX11Shader

```cpp
namespace fbzz::renderer {

class DX11Shader : public IShader {
public:
    bool Init(ID3D11Device* device, const std::string& path);

    void SetConstantBuffer(uint32_t slot, const void* data, size_t sizeBytes) override;
    void SetTexture(uint32_t slot, std::shared_ptr<ITexture> texture) override;

    void Bind(ID3D11DeviceContext* ctx) const;

private:
    Microsoft::WRL::ComPtr<ID3D11VertexShader>  m_vs;
    Microsoft::WRL::ComPtr<ID3D11PixelShader>   m_ps;
    Microsoft::WRL::ComPtr<ID3D11InputLayout>   m_inputLayout;

    // スロットごとの定数バッファ
    std::array<Microsoft::WRL::ComPtr<ID3D11Buffer>, 4> m_cbuffers;

    ID3D11Device*        m_device  = nullptr;
    ID3D11DeviceContext* m_context = nullptr;
};

} // namespace fbzz::renderer
```
